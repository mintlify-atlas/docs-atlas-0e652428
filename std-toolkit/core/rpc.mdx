## Overview

The RPC utilities automatically generate type-safe CRUD endpoints from your ESchema definitions. They integrate with [@effect/rpc](https://effect.website/docs/rpc/introduction) for building RPC servers.

## API

### makeEntityRpcGroup

Generate all four CRUD operations at once:

```typescript
function makeEntityRpcGroup<N, Id, V, F, P>(
  eschema: ESchema<N, Id, V, F>,
  prefix?: P
): [
  GetRpc<N, Id, V, F, P>,
  InsertRpc<N, Id, V, F, P>,
  UpdateRpc<N, Id, V, F, P>,
  DeleteRpc<N, Id, V, F, P>
]
```

<ParamField path="eschema" type="ESchema" required>
  The ESchema instance to generate RPCs for
</ParamField>

<ParamField path="prefix" type="string">
  Optional prefix for RPC names (e.g., "admin/")
</ParamField>

**Returns:** Array of `[GetRpc, InsertRpc, UpdateRpc, DeleteRpc]`

#### Example

```typescript
import { ESchema } from "@std-toolkit/eschema";
import { makeEntityRpcGroup } from "@std-toolkit/core/rpc";
import { Schema } from "effect";

const UserSchema = ESchema.make("User", "userId", {
  name: Schema.String,
  email: Schema.String,
}).build();

const [GetUser, InsertUser, UpdateUser, DeleteUser] = makeEntityRpcGroup(
  UserSchema
);

// RPC names: "getUser", "insertUser", "updateUser", "deleteUser"
```

### makeGetRpc

Generate a Get RPC for fetching a single entity:

```typescript
function makeGetRpc<N, Id, V, F, P>(
  eschema: ESchema<N, Id, V, F>,
  prefix?: P
): Rpc<`${P}get${N}`, Payload, EntityType<T> | null, StdToolkitError>
```

**Payload:** `{ [idField]: string }`

**Success:** `EntityType<T> | null` (null if not found)

**Error:** `StdToolkitError`

#### Example

```typescript
const GetUser = makeGetRpc(UserSchema);

// Handler
const handler = GetUser.handler((payload) =>
  Effect.gen(function* () {
    const user = yield* Users.get({ userId: payload.userId });
    return user; // Can be null
  })
);

// Client usage
const user = yield* GetUser({ userId: "user-123" });
if (user) {
  console.log(user.value.name);
}
```

### makeInsertRpc

Generate an Insert RPC for creating entities:

```typescript
function makeInsertRpc<N, Id, V, F, P>(
  eschema: ESchema<N, Id, V, F>,
  prefix?: P
): Rpc<`${P}insert${N}`, Payload, EntityType<T>, StdToolkitError>
```

**Payload:** Entity fields **excluding** the ID field

**Success:** `EntityType<T>` (created entity with metadata)

**Error:** `StdToolkitError`

#### Example

```typescript
const InsertUser = makeInsertRpc(UserSchema);

// Handler
const handler = InsertUser.handler((payload) =>
  Users.insert(payload)  // ID is auto-generated
);

// Client usage
const user = yield* InsertUser({
  name: "Alice",
  email: "alice@example.com",
});

console.log(user.value.userId); // "01HQXYZ..."
console.log(user.meta._uid);    // "01HQABC..."
```

<Note>
  The ID field is automatically excluded from the payload and will be auto-generated by the entity.
</Note>

---

### makeUpdateRpc

Generate an Update RPC for modifying entities:

```typescript
function makeUpdateRpc<N, Id, V, F, P>(
  eschema: ESchema<N, Id, V, F>,
  prefix?: P
): Rpc<`${P}update${N}`, Payload, EntityType<T>, StdToolkitError>
```

**Payload:** `{ [idField]: string; updates: Partial<Fields> }`

**Success:** `EntityType<T>` (updated entity with metadata)

**Error:** `StdToolkitError`

#### Example

```typescript
const UpdateUser = makeUpdateRpc(UserSchema);

// Handler
const handler = UpdateUser.handler((payload) =>
  Users.update(
    { userId: payload.userId },
    payload.updates
  )
);

// Client usage
const updated = yield* UpdateUser({
  userId: "user-123",
  updates: {
    name: "Bob",
  },
});

console.log(updated.value.name);  // "Bob"
console.log(updated.value.email); // Unchanged
```

---

### makeDeleteRpc

Generate a Delete RPC for soft-deleting entities:

```typescript
function makeDeleteRpc<N, Id, V, F, P>(
  eschema: ESchema<N, Id, V, F>,
  prefix?: P
): Rpc<`${P}delete${N}`, Payload, EntityType<T>, StdToolkitError>
```

**Payload:** `{ [idField]: string }`

**Success:** `EntityType<T>` (deleted entity with `_d: true`)

**Error:** `StdToolkitError`

#### Example

```typescript
const DeleteUser = makeDeleteRpc(UserSchema);

// Handler
const handler = DeleteUser.handler((payload) =>
  Users.delete({ userId: payload.userId })
);

// Client usage
const deleted = yield* DeleteUser({ userId: "user-123" });

console.log(deleted.meta._d); // true
```

<Warning>
  Deletes are **soft deletes**. The entity is marked with `_d: true` but remains in the database.
</Warning>

### StdToolkitError

Standard error type for all RPC operations:

```typescript
class StdToolkitError extends Schema.TaggedError<StdToolkitError>()("StdToolkitError", {
  message: Schema.String,
  code: Schema.optional(Schema.String),
})
```

<ResponseField name="message" type="string" required>
  Human-readable error message
</ResponseField>

<ResponseField name="code" type="string">
  Optional error code (e.g., "NOT_FOUND", "VALIDATION_ERROR")
</ResponseField>

#### Example

```typescript
import { StdToolkitError } from "@std-toolkit/core/rpc";
import { Effect } from "effect";

const handler = GetUser.handler((payload) =>
  Effect.gen(function* () {
    const user = yield* Users.get({ userId: payload.userId });
    
    if (!user) {
      return yield* new StdToolkitError({
        message: `User ${payload.userId} not found`,
        code: "NOT_FOUND",
      });
    }
    
    return user;
  })
);

// Client error handling
const result = yield* GetUser({ userId: "invalid" }).pipe(
  Effect.catchTag("StdToolkitError", (error) => {
    console.log(error.message); // "User invalid not found"
    console.log(error.code);    // "NOT_FOUND"
    return Effect.succeed(null);
  })
);
```

## Complete Example

### Server

```typescript server.ts
import { Rpc } from "@effect/rpc";
import { HttpServer } from "@effect/platform";
import { NodeHttpServer } from "@effect/platform-node";
import { makeEntityRpcGroup } from "@std-toolkit/core/rpc";
import { Effect, Layer } from "effect";
import Database from "better-sqlite3";
import { SqliteDBBetterSqlite3 } from "@std-toolkit/sqlite/adapters/better-sqlite3";

// Define schema and entity (from previous examples)
const [GetUser, InsertUser, UpdateUser, DeleteUser] = makeEntityRpcGroup(
  UserSchema
);

// Create RPC router
const router = Rpc.router({
  getUser: GetUser.handler((payload) =>
    Users.get({ userId: payload.userId })
  ),
  
  insertUser: InsertUser.handler((payload) =>
    Users.insert(payload)
  ),
  
  updateUser: UpdateUser.handler((payload) =>
    Users.update({ userId: payload.userId }, payload.updates)
  ),
  
  deleteUser: DeleteUser.handler((payload) =>
    Users.delete({ userId: payload.userId })
  ),
});

// Create HTTP server
const db = new Database("app.db");
const dbLayer = SqliteDBBetterSqlite3(db);

const server = HttpServer.router.empty.pipe(
  HttpServer.router.post("/rpc", Rpc.toHttpApp(router))
);

const ServerLive = NodeHttpServer.layer({ port: 3000 });

const program = Effect.gen(function* () {
  yield* Users.setup();
  yield* HttpServer.serve(server);
});

Effect.runPromise(
  program.pipe(
    Effect.provide(dbLayer),
    Effect.provide(ServerLive)
  )
);
```

### Client

```typescript client.ts
import { Rpc } from "@effect/rpc";
import { HttpClient } from "@effect/platform";
import { makeEntityRpcGroup } from "@std-toolkit/core/rpc";
import { Effect } from "effect";

const [GetUser, InsertUser, UpdateUser, DeleteUser] = makeEntityRpcGroup(
  UserSchema
);

const client = Rpc.client(
  HttpClient.fetch.pipe(
    HttpClient.mapRequest(
      HttpClient.request.post("http://localhost:3000/rpc")
    )
  )
);

const program = Effect.gen(function* () {
  // Insert
  const user = yield* client(InsertUser, {
    name: "Alice",
    email: "alice@example.com",
  });
  
  console.log("Created:", user.value.userId);
  
  // Get
  const fetched = yield* client(GetUser, {
    userId: user.value.userId,
  });
  
  console.log("Fetched:", fetched?.value.name);
  
  // Update
  const updated = yield* client(UpdateUser, {
    userId: user.value.userId,
    updates: { name: "Bob" },
  });
  
  console.log("Updated:", updated.value.name);
  
  // Delete
  const deleted = yield* client(DeleteUser, {
    userId: user.value.userId,
  });
  
  console.log("Deleted:", deleted.meta._d); // true
});

Effect.runPromise(program);
```

## Type Safety

RPC payloads and responses are fully typed:

```typescript
// InsertUser payload type is inferred
const user = yield* InsertUser({
  name: "Alice",
  email: "alice@example.com",
  // userId is excluded (auto-generated)
});

// Response type is EntityType<User>
user.value.userId;  // string
user.value.name;    // string
user.meta._uid;     // string

// Compile error: Property 'age' does not exist
const invalid = yield* InsertUser({
  name: "Alice",
  age: 30,  // ❌ Error
});
```

## Error Handling

```typescript
import { Effect } from "effect";

const program = Effect.gen(function* () {
  const user = yield* GetUser({ userId: "invalid" });
  return user;
}).pipe(
  Effect.catchTag("StdToolkitError", (error) => {
    console.error("RPC Error:", error.message);
    if (error.code === "NOT_FOUND") {
      // Handle not found
    }
    return Effect.succeed(null);
  }),
  Effect.catchTag("SqliteDBError", (error) => {
    console.error("Database Error:", error.error._tag);
    return Effect.succeed(null);
  })
);
```

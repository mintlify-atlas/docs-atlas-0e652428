## Basic Setup

### 1. Define Your Schema

Start by creating an ESchema for your entity:

```typescript
import { ESchema } from "@std-toolkit/eschema";
import { Schema } from "effect";

const UserSchema = ESchema.make("User", "userId", {
  email: Schema.String,
  name: Schema.String,
  status: Schema.Literal("active", "inactive"),
}).build();
```

### 2. Create a Table

Define the single table structure:

```typescript
import { SQLiteTable } from "@std-toolkit/sqlite";

const table = SQLiteTable.make({ tableName: "app_data" })
  .primary("pk", "sk")  // Primary key columns
  .index("IDX1", "IDX1PK", "IDX1SK")  // Secondary index
  .build();
```

<Note>
  The primary key is **always** `(pk, sk)`. Secondary indexes use separate columns like `IDX1PK` and `IDX1SK`.
</Note>

### 3. Create an Entity

Configure how the entity maps to the table:

```typescript
import { SQLiteEntity } from "@std-toolkit/sqlite";

const Users = SQLiteEntity.make(table)
  .eschema(UserSchema)  // Use the schema
  .primary()  // pk: "User", sk: userId
  .index("IDX1", "byEmail", { pk: ["email"] })  // pk: "byEmail#{email}", sk: _uid
  .build();
```

**How indexes are derived:**

- **Primary**: `pk = "User"`, `sk = userId` (from idField)
- **byEmail**: `pk = "byEmail#{email}"`, `sk = _uid`

### 4. Create a Database Layer

Provide a database implementation:

<CodeGroup>

```typescript Node.js
import Database from "better-sqlite3";
import { SqliteDBBetterSqlite3 } from "@std-toolkit/sqlite/adapters/better-sqlite3";

const db = new Database("app.db");
const layer = SqliteDBBetterSqlite3(db);
```

```typescript Cloudflare DO
import { SqliteDBDO } from "@std-toolkit/sqlite/adapters/do";

export class MyDurableObject {
  constructor(state, env) {
    this.layer = SqliteDBDO(state.storage.sql);
  }
}
```

</CodeGroup>

### 5. Use It

Perform database operations:

```typescript
import { Effect } from "effect";

const program = Effect.gen(function* () {
  // Setup table
  yield* Users.setup();
  
  // Insert a user
  const user = yield* Users.insert({
    userId: "user-1",
    email: "alice@example.com",
    name: "Alice",
    status: "active",
  });
  
  console.log("Created:", user.value);
  
  // Get by ID
  const found = yield* Users.get({ userId: "user-1" });
  
  if (found) {
    console.log("Found:", found.value.name);
  }
  
  // Query by email
  const byEmail = yield* Users.query("byEmail", {
    pk: { email: "alice@example.com" },
    sk: { ">=": null },
  });
  
  console.log("By email:", byEmail.items.length);
});

Effect.runPromise(program.pipe(Effect.provide(layer)));
```

## Complete Example

```typescript app.ts
import { ESchema } from "@std-toolkit/eschema";
import { SQLiteTable, SQLiteEntity } from "@std-toolkit/sqlite";
import { Schema, Effect } from "effect";
import Database from "better-sqlite3";
import { SqliteDBBetterSqlite3 } from "@std-toolkit/sqlite/adapters/better-sqlite3";

// 1. Define schema
const UserSchema = ESchema.make("User", "userId", {
  email: Schema.String,
  name: Schema.String,
  status: Schema.Literal("active", "inactive"),
}).build();

// 2. Create table
const table = SQLiteTable.make({ tableName: "app_data" })
  .primary("pk", "sk")
  .index("IDX1", "IDX1PK", "IDX1SK")
  .build();

// 3. Create entity
const Users = SQLiteEntity.make(table)
  .eschema(UserSchema)
  .primary()
  .index("IDX1", "byEmail", { pk: ["email"] })
  .build();

// 4. Create database layer
const db = new Database("app.db");
const layer = SqliteDBBetterSqlite3(db);

// 5. Run the program
const program = Effect.gen(function* () {
  // Setup
  yield* Users.setup();
  
  // Insert
  const alice = yield* Users.insert({
    userId: "user-1",
    email: "alice@example.com",
    name: "Alice",
    status: "active",
  });
  console.log("Created Alice:", alice.value.userId);
  
  const bob = yield* Users.insert({
    userId: "user-2",
    email: "bob@example.com",
    name: "Bob",
    status: "inactive",
  });
  console.log("Created Bob:", bob.value.userId);
  
  // Get by ID
  const user = yield* Users.get({ userId: "user-1" });
  console.log("Found:", user?.value.name);
  
  // Query all users
  const allUsers = yield* Users.query("primary", {
    pk: {},
    sk: { ">=": null },
  });
  console.log("Total users:", allUsers.items.length);
  
  // Query by email
  const byEmail = yield* Users.query("byEmail", {
    pk: { email: "alice@example.com" },
    sk: { ">=": null },
  });
  console.log("Alice by email:", byEmail.items[0]?.value.name);
  
  // Update
  const updated = yield* Users.update(
    { userId: "user-1" },
    { status: "inactive" }
  );
  console.log("Updated status:", updated.value.status);
  
  // Delete (soft delete)
  const deleted = yield* Users.delete({ userId: "user-2" });
  console.log("Deleted:", deleted.meta._d);  // true
});

Effect.runPromise(program.pipe(Effect.provide(layer)));
```

## Multi-Entity Example

Store multiple entity types in one table:

```typescript
import { ESchema } from "@std-toolkit/eschema";
import { SQLiteTable, SQLiteEntity, EntityRegistry } from "@std-toolkit/sqlite";
import { Schema, Effect } from "effect";
import Database from "better-sqlite3";
import { SqliteDBBetterSqlite3 } from "@std-toolkit/sqlite/adapters/better-sqlite3";

// Schemas
const UserSchema = ESchema.make("User", "userId", {
  name: Schema.String,
  email: Schema.String,
}).build();

const PostSchema = ESchema.make("Post", "postId", {
  authorId: Schema.String,
  title: Schema.String,
  content: Schema.String,
}).build();

// Single table
const table = SQLiteTable.make({ tableName: "app_data" })
  .primary("pk", "sk")
  .index("IDX1", "IDX1PK", "IDX1SK")
  .build();

// Entities
const Users = SQLiteEntity.make(table)
  .eschema(UserSchema)
  .primary()  // pk: "User", sk: userId
  .index("IDX1", "byEmail", { pk: ["email"] })
  .build();

const Posts = SQLiteEntity.make(table)
  .eschema(PostSchema)
  .primary({ pk: ["authorId"] })  // pk: "Post#{authorId}", sk: postId
  .build();

// Registry for setup
const registry = EntityRegistry.make(table)
  .register(Users)
  .register(Posts)
  .build();

// Database layer
const db = new Database("app.db");
const layer = SqliteDBBetterSqlite3(db);

// Program
const program = Effect.gen(function* () {
  // Setup both entities
  yield* registry.setup();
  
  // Insert user
  const user = yield* Users.insert({
    userId: "user-1",
    name: "Alice",
    email: "alice@example.com",
  });
  
  // Insert posts by this user
  const post1 = yield* Posts.insert({
    authorId: user.value.userId,
    postId: "post-1",
    title: "First Post",
    content: "Hello World",
  });
  
  const post2 = yield* Posts.insert({
    authorId: user.value.userId,
    postId: "post-2",
    title: "Second Post",
    content: "More content",
  });
  
  // Query all posts by author
  const authorPosts = yield* Posts.query("primary", {
    pk: { authorId: user.value.userId },
    sk: { ">=": null },
  });
  
  console.log(`Alice has ${authorPosts.items.length} posts`);
});

Effect.runPromise(program.pipe(Effect.provide(layer)));
```

## Understanding pk and sk

### Primary Index

When you define `.primary()` or `.primary({ pk: [...] })`:

```typescript
// No pk fields: pk is just the entity name
Users.primary()
// pk: "User"
// sk: userId (from idField)

// With pk fields: pk includes field values
Posts.primary({ pk: ["authorId"] })
// pk: "Post#{authorId}"
// sk: postId (from idField)
```

### Secondary Indexes

When you define `.index(tableName, entityName, { pk: [...] })`:

```typescript
Users.index("IDX1", "byEmail", { pk: ["email"] })
// IDX1PK: "byEmail#{email}"
// IDX1SK: _uid (ULID for ordering)

Posts.index("IDX1", "byStatus", { pk: ["status"] })
// IDX1PK: "byStatus#{status}"
// IDX1SK: _uid
```

<Note>
  Secondary indexes always use `_uid` as the sort key for time-ordering.
</Note>

## Auto-Generated IDs

If you don't provide an ID field, it's auto-generated:

```typescript
const user = yield* Users.insert({
  // userId is auto-generated as a ULID
  email: "alice@example.com",
  name: "Alice",
  status: "active",
});

console.log(user.value.userId);  // "01HQXYZ123ABC..."
```

## Metadata Access

Every operation returns both the value and metadata:

```typescript
const user = yield* Users.insert({
  userId: "user-1",
  email: "alice@example.com",
  name: "Alice",
  status: "active",
});

// Access value
console.log(user.value.userId);     // "user-1"
console.log(user.value.name);       // "Alice"

// Access metadata
console.log(user.meta._v);          // "v1" (schema version)
console.log(user.meta._uid);        // "01HQABC..." (update ID)
console.log(user.meta._d);          // false (not deleted)
console.log(user.meta._e);          // "User" (entity name)
```

## Error Handling

```typescript
import { SqliteDBError } from "@std-toolkit/sqlite";

const program = Effect.gen(function* () {
  const user = yield* Users.get({ userId: "invalid" });
  return user;
}).pipe(
  Effect.catchTag("SqliteDBError", (error) => {
    console.error("Error type:", error.error._tag);
    console.error("Cause:", error.error.cause);
    return Effect.succeed(null);
  })
);
```

## Testing

Use in-memory databases for tests:

```typescript
import { describe, it, expect, beforeEach } from "@effect/vitest";
import Database from "better-sqlite3";

describe("Users", () => {
  let db: Database.Database;
  let layer: Layer.Layer<SqliteDB>;

  beforeEach(() => {
    db = new Database(":memory:");
    layer = SqliteDBBetterSqlite3(db);
  });

  it.effect("creates a user", () =>
    Effect.gen(function* () {
      yield* Users.setup();
      
      const user = yield* Users.insert({
        userId: "user-1",
        email: "test@example.com",
        name: "Test",
        status: "active",
      });

      expect(user.value.userId).toBe("user-1");
      expect(user.value.name).toBe("Test");
    }).pipe(Effect.provide(layer))
  );
});
```
